185// Returns a random number in [l, r].
186template <typename T> T next(T l, T r) {
187 tgen_ensure(l <= r, "range for `next` bust be valid");
188 if constexpr (std::is_integral_v<T>)
189 return std::uniform_int_distribution<T>(l, r)(rng_internal);
190 else if constexpr (std::is_floating_point_v<T>)
191 return std::uniform_real_distribution<T>(l, r)(rng_internal);
192 else
193 throw error_internal("invalid type for next (" +
194 std::string(typeid(T).name()) + ")");
195}

197// Shuffles [first, last) inplace uniformly, for RandomAccessIterator.
198template <typename It> void shuffle(It first, It last) {
199 if (first == last)
200 return;
201
202 for (It i = first + 1; i != last; ++i)
203 std::iter_swap(i, first + next(0, static_cast<int>(i - first)));
204}

249// Chooses k values from the container, as in a subsequence of size k. Returns a
250// copy.
251template <typename C> C choose(int k, const C &container) {
252 tgen_ensure(0 < k and k <= container.size(),
253 "number of elements to choose must be valid");
254 std::vector<typename C::value_type> new_vec;
255 C new_container;
256 int need = k, left = container.size();
257 for (auto cur_it = container.begin(); cur_it != container.end(); ++cur_it) {
258 if (next(1, left--) <= need) {
259 new_container.insert(new_container.end(), *cur_it);
260 need--;
261 }
262 }
263 return new_container;
264}

307/*
308 * Opts - options given to the generator.
309 *
310 * Incompatible with testlib.
311 *
312 * Opts are a list of either positional or named options.
313 *
314 * Named options is given in one of the following formats:
315 * 1) -keyname=value or --keyname=value (ex. -n=10 , --test-count=20)
316 * 2) -keyname value or --keyname value (ex. -n 10 , --test-count 20)
317 *
318 * Positional options are number from 0 sequentially.
319 * For example, for "10 -n=20 str" positional option 1 is the string "str".
320 */
321
322inline std::vector<std::string>
323 pos_opts_internal; // Dictionary containing the positional parsed opts.
324inline std::map<std::string, std::string>
325 named_opts_internal; // Global dictionary the named parsed opts.
326
327// Returns true if there is an opt at a given index.
328inline bool has_opt(std::size_t index) {
329 return 0 <= index and index < pos_opts_internal.size();
330}
331
332// Returns true if there is an opt with a given key.
333inline bool has_opt(const std::string &key) {
334 return named_opts_internal.count(key) != 0;
335}
336
337template <typename T> T get_opt_internal(const std::string &value) {
338 try {
339 if constexpr (std::is_same_v<T, bool>) {
340 if (value == "true" or value == "1")
341 return true;
342 if (value == "false" or value == "0")
343 return false;
344 } else if constexpr (std::is_integral_v<T>) {
345 if constexpr (std::is_unsigned_v<T>)
346 return static_cast<T>(std::stoull(value));
347 else
348 return static_cast<T>(std::stoll(value));
349 } else if constexpr (std::is_floating_point_v<T>)
350 return static_cast<T>(std::stold(value));
351 else
352 return value; // default: std::string
353 } catch (...) {
354 }
355
356 throw error_internal("invalid value `" + value + "` for type " +
357 typeid(T).name());
358}
359
360// Returns the parsed opt by a given key. If no opts with the given key are
361// found, returns the given default_value.
362template <typename T, typename KEY>
363T opt(const KEY &key, std::optional<T> default_value = std::nullopt) {
364 if constexpr (std::is_same_v<KEY, int>) {
365 if (!has_opt(key)) {
366 if (default_value)
367 return *default_value;
368 throw error_internal("cannot find key with index " +
369 std::to_string(key));
370 }
371 return get_opt_internal<T>(pos_opts_internal[key]);
372 } else { // std::string
373 if (!has_opt(key)) {
374 if (default_value)
375 return *default_value;
376 throw error_internal("cannot find key with key " +
377 std::string(key));
378 }
379 return get_opt_internal<T>(named_opts_internal[key]);
380 }
381}

383inline void parse_opts_internal(int argc, char **argv) {
384 // Parses the opts into `pos_opts_internal` vector and `named_opts_internal`
385 // map. Starting from 1 to ignore the name of the executable.
386 for (int i = 1; i < argc; i++) {
387 std::string key(argv[i]);
388
389 if (key[0] == '-') {
390 tgen_ensure(key.size() > 1,
391 "invalid opt (" + std::string(argv[i]) + ")");
392 if ('0' <= key[1] and key[1] <= '9') {
393 // This case is a positional negative number argument
394 pos_opts_internal.push_back(key);
395 continue;
396 }
397
398 // pops first char '-'
399 key = key.substr(1);
400 } else {
401 // This case is a positional argument that does not start with '-'
402 pos_opts_internal.push_back(key);
403 continue;
404 }
405
406 // Pops a possible second char '-'.
407 if (key[0] == '-') {
408 tgen_ensure(key.size() > 1,
409 "invalid opt (" + std::string(argv[i]) + ")");
410
411 // pops first char '-'
412 key = key.substr(1);
413 }
414
415 // Assumes that, if it starts with '-' and second char is not a digit,
416 // then it is a <key, value> pair.
417 // 1 or 2 chars '-' have already been poped.
418
419 std::size_t eq = key.find('=');
420 if (eq != std::string::npos) {
421 // This is the '--key=value' case.
422 std::string value = key.substr(eq + 1);
423 key = key.substr(0, eq);
424 tgen_ensure(!key.empty() and !value.empty(),
425 "expected non-empty key/value in opt (" +
426 std::string(argv[1]));
427 tgen_ensure(named_opts_internal.count(key) == 0,
428 "cannot have repeated keys");
429 named_opts_internal[key] = value;
430 } else {
431 // This is the '--key value' case.
432 tgen_ensure(named_opts_internal.count(key) == 0,
433 "cannot have repeated keys");
434 tgen_ensure(argv[i + 1], "value cannot be empty");
435 named_opts_internal[key] = std::string(argv[i + 1]);
436 i++;
437 }
438 }
439}
440inline void set_seed_internal(int argc, char **argv) {
441 std::vector<uint32_t> seed;
442
443 // Starting from 1 to ignore the name of the executable.
444 for (int i = 1; i < argc; ++i) {
445 // We append the number of chars, and then the list of chars.
446 int size_pos = seed.size();
447 seed.push_back(0);
448 for (char *s = argv[i]; *s != '\0'; ++s) {
449 ++seed[size_pos];
450 seed.push_back(*s);
451 }
452 }
453 std::seed_seq seq(seed.begin(), seed.end());
454 rng_internal.seed(seq);
455}

458inline void register_gen(int argc, char **argv) {
459 set_seed_internal(argc, argv);
460
461 pos_opts_internal.clear();
462 named_opts_internal.clear();
463 parse_opts_internal(argc, argv);
464}

476template <typename T> struct sequence : gen_base<sequence<T>> {
477 int size_; // Size of sequence.
478 T value_l_, value_r_; // Range of defined values.
479 std::set<T> values_; // Set of values. If empty, use range. if not,
480 // represents the possible values, and the range
481 // represents the index in this set)
482 std::map<T, int>
483 value_idx_in_set_; // Index of every value in the set above.
484 std::vector<std::pair<T, T>> val_range_; // Range of values of each index.
485 std::vector<std::vector<int>> neigh_; // Adjacency list of equality.
486 std::vector<std::set<int>>
487 distinct_constraints_; // All distinct constraints.
488
489 // Creates generator for sequences of size 'size', with random T in [l, r].
490 sequence(int size, T value_l, T value_r)
491 : size_(size), value_l_(value_l), value_r_(value_r), neigh_(size) {
492 tgen_ensure(size_ > 0, "size must be positive");
493 tgen_ensure(value_l_ <= value_r_, "value range must be valid");
494 for (int i = 0; i < size_; ++i)
495 val_range_.emplace_back(value_l_, value_r_);
496 }
497
498 // Creates sequence with value set.
499 sequence(int size, std::set<T> values)
500 : size_(size), values_(values), neigh_(size) {
501 tgen_ensure(size_ > 0, "size must be positive");
502 tgen_ensure(!values.empty(), "value set must be non-empty");
503 value_l_ = 0, value_r_ = values.size() - 1;
504 for (int i = 0; i < size_; ++i)
505 val_range_.emplace_back(value_l_, value_r_);
506 int idx = 0;
507 for (T value : values_)
508 value_idx_in_set_[value] = idx++;
509 }
510
511 // Restricts sequences for sequence[idx] = value.
512 sequence &set(int idx, T value) {
513 tgen_ensure(0 <= idx and idx < size_, "index must be valid");
514 if (values_.size() == 0) {
515 auto &[left, right] = val_range_[idx];
516 if (left == right and value_l_ != value_r_) {
517 tgen_ensure(left == value,
518 "must not set to two different values");
519 } else {
520 tgen_ensure(left <= value and value <= right,
521 "value must be in the defined range");
522 }
523 left = right = value;
524 } else {
525 tgen_ensure(values_.count(value),
526 "value must be in the set of values");
527 auto &[left, right] = val_range_[idx];
528 int new_val = value_idx_in_set_[value];
529 tgen_ensure(left <= new_val and new_val <= right,
530 "must not set to two different values");
531 left = right = new_val;
532 }
533 return *this;
534 }
535
536 // Restricts sequences for sequence[idx_1] = sequence[idx_2].
537 sequence &equal(int idx_1, int idx_2) {
538 tgen_ensure(0 <= std::min(idx_1, idx_2) and
539 std::max(idx_1, idx_2) < size_,
540 "indices must be valid");
541 if (idx_1 == idx_2)
542 return *this;
543
544 neigh_[idx_1].push_back(idx_2);
545 neigh_[idx_2].push_back(idx_1);
546 return *this;
547 }
548
549 // Restricts sequences for sequence[left..right] to have all equal values.
550 sequence &equal_range(int left, int right) {
551 tgen_ensure(0 <= left and left <= right and right < size_,
552 "range indices bust be valid");
553 for (int i = left; i < right; ++i)
554 equal(i, i + 1);
555 return *this;
556 }
557
558 // Restricts sequences for sequence[S] to be distinct, for given subset S of
559 // indices.
560 // You can not add two of these restrictions with intersection.
561 sequence &distinct(std::set<int> indices) {
562 if (!indices.empty())
563 distinct_constraints_.push_back(indices);
564 return *this;
565 }
566
567 // Restricts sequences for sequence[idx_1] != sequence[idx_2].
568 sequence &different(int idx_1, int idx_2) {
569 std::set<int> indices = {idx_1, idx_2};
570 return distinct(indices);
571 }
572
573 // Restricts sequences with distinct elements.
574 sequence &distinct() {
575 std::set<int> indices;
576 for (int i = 0; i < size_; ++i)
577 indices.insert(i);
578 return distinct(indices);
579 }

639 // Generates a uniformly random list of k distinct values in `[value_l,
640 // value_r]`, such that no value is in `forbidden_values`.
641 std::vector<T>
642 generate_distinct_values(int k, const std::set<T> &forbidden_values) {
643 for (auto forbidden : forbidden_values)
644 tgen_ensure(value_l_ <= forbidden and forbidden <= value_r_);
645 // We generate our numbers in the range [0, num_available) with
646 // num_available = (r-l+1)-(forbidden_values.size()), and then map them
647 // to the correct range. We will run k steps of Fisher–Yates, using a
648 // map to store a virtual sequence that starts with a[i] = i.
649 T num_available = (value_r_ - value_l_ + 1) - forbidden_values.size();
650 if (num_available < k)
651 throw error_internal(
652 "failed to generate sequence: complex constraints");
653 std::map<T, T> virtual_list;
654 std::vector<T> gen_list;
655 for (int i = 0; i < k; i++) {
656 T j = next<T>(i, num_available - 1);
657 T vj = virtual_list.count(j) ? virtual_list[j] : j;
658 T vi = virtual_list.count(i) ? virtual_list[i] : i;
659
660 virtual_list[j] = vi, virtual_list[i] = vj;
661
662 gen_list.push_back(virtual_list[i]);
663 }
664
665 // Shifts back to correct range, but there might still be values
666 // that we can not use.
667 for (T &value : gen_list)
668 value += value_l_;
669
670 // Now for every generated value, we shift it by how many forbidden
671 // values are <= to it.
672 std::vector<std::pair<T, int>> values_sorted;
673 for (std::size_t i = 0; i < gen_list.size(); ++i)
674 values_sorted.emplace_back(gen_list[i], i);
675 // We iterate through them in increasing order.
676 std::sort(values_sorted.begin(), values_sorted.end());
677 auto cur_it = forbidden_values.begin();
678 int smaller_forbidden_count = 0;
679 for (auto [val, idx] : values_sorted) {
680 while (cur_it != forbidden_values.end() and
681 *cur_it <= val + smaller_forbidden_count)
682 ++cur_it, ++smaller_forbidden_count;
683 gen_list[idx] += smaller_forbidden_count;
684 }
685
686 return gen_list;
687 }

689 // Generates sequence instance.
690 instance gen() {
691 std::vector<T> vec(size_);
692 std::vector<bool> defined_idx(
693 size_, false); // For every index, if it has been set in `vec`.
694
695 std::vector<int> comp_id(size_, -1); // Component id of each index.
696 std::vector<std::vector<int>> comp(size_); // Component of each comp-id.
697 int comp_count = 0; // Number of different components.
698
699 // Defines value of entire component.
700 auto define_comp = [&](int cur_comp, T val) {
701 for (int idx : comp[cur_comp]) {
702 tgen_ensure(!defined_idx[idx]);
703 vec[idx] = val;
704 defined_idx[idx] = true;
705 }
706 };
707
708 // Groups = components.
709 {
710 std::vector<bool> vis(size_, false); // Visited for each index.
711 for (int idx = 0; idx < size_; ++idx)
712 if (!vis[idx]) {
713 T new_value;
714 bool value_defined = false;
715
716 // BFS to visit the connected component, grouping equal
717 // values.
718 std::queue<int> q({idx});
719 vis[idx] = true;
720 std::vector<int> component;
721 while (!q.empty()) {
722 int cur_idx = q.front();
723 q.pop();
724
725 component.push_back(cur_idx);
726
727 // Checks value.
728 auto [l, r] = val_range_[cur_idx];
729 if (l == r) {
730 if (!value_defined) {
731 // We found the value.
732 value_defined = true;
733 new_value = l;
734 } else if (new_value != l) {
735 // We found a contradiction
736 throw contradiction_error_internal(
737 "sequence",
738 "tried to set value to `" +
739 std::to_string(new_value) +
740 "`, but it was already set as `" +
741 std::to_string(l) + "`");
742 }
743 }
744
745 for (int nxt_idx : neigh_[cur_idx]) {
746 if (!vis[nxt_idx]) {
747 vis[nxt_idx] = true;
748 q.push(nxt_idx);
749 }
750 }
751 }
752
753 // Group entire component, checking if value is defined.
754 for (int cur_idx : component) {
755 comp_id[cur_idx] = comp_count;
756 comp[comp_id[cur_idx]].push_back(cur_idx);
757 }
758
759 // Sets value if needed.
760 if (value_defined)
761 define_comp(comp_count, new_value);
762
763 ++comp_count;
764 }
765 }
766
767 // Initial parsing of distinct constraints.
768 std::vector<std::set<int>> distinct_containing_comp_idx(comp_count);
769 {
770 int dist_id = 0;
771 for (const std::set<int> &distinct : distinct_constraints_) {
772 // Checks if there are too many distinct values.
773 if (static_cast<uint64_t>(distinct.size() - 1) +
774 static_cast<uint64_t>(value_l_) >
775 static_cast<uint64_t>(value_r_))
776 throw contradiction_error_internal(
777 "sequence",
778 "tried to generate " + std::to_string(distinct.size()) +
779 " distinct values, but the maximum is " +
780 std::to_string(value_r_ - value_l_ + 1));
781
782 // Checks if two values in same component are marked as
783 // different.
784 std::set<int> comp_ids;
785 for (int idx : distinct) {
786 if (comp_ids.count(comp_id[idx]))
787 throw contradiction_error_internal(
788 "sequence", "tried to set two indices as equal and "
789 "different");
790 comp_ids.insert(comp_id[idx]);
791
792 distinct_containing_comp_idx[comp_id[idx]].insert(dist_id);
793 }
794 ++dist_id;
795 }
796 }
797
798 // If some value is in >= 3 sets, then there is a cycle.
799 for (auto &distinct_containing : distinct_containing_comp_idx)
800 if (distinct_containing.size() >= 3)
801 throw error_internal(
802 "failed to generate sequence: complex constraints");
803
804 std::vector<bool> vis_distinct(distinct_constraints_.size(), false);
805 std::vector<bool> initially_defined_comp_idx(comp_count, false);
806
807 // Fills the value in a tree defined by distinct constraints.
808 auto define_tree = [&](int distinct_id) {
809 // The set `distinct_constraints_[distinct_id]` can have some values
810 // that are defined.
811
812 // Generates set of already defined values.
813 std::set<T> defined_values;
814 for (int idx : distinct_constraints_[distinct_id])
815 if (defined_idx[idx]) {
816 // Checks if two values in `distinct_constraints_[dist_id]`
817 // have been set to the same value
818 if (defined_values.count(vec[idx]))
819 throw contradiction_error_internal(
820 "sequence",
821 "tried to set two indices as equal and different");
822
823 defined_values.insert(vec[idx]);
824 }
825
826 // Generates values in this root distinct constraint.
827 {
828 int new_value_count =
829 distinct_constraints_[distinct_id].size() -
830 static_cast<int>(defined_values.size());
831 std::vector<T> generated_values =
832 generate_distinct_values(new_value_count, defined_values);
833 auto val_it = generated_values.begin();
834 for (int idx : distinct_constraints_[distinct_id])
835 if (defined_idx[idx]) {
836 // The root can cover these components, but there should
837 // not be any other defined in this tree.
838 initially_defined_comp_idx[comp_id[idx]] = false;
839 } else {
840 define_comp(comp_id[idx], *val_it);
841 ++val_it;
842 }
843 }
844
845 // BFS on the tree of distinct constraints.
846 std::queue<std::pair<int, int>> q; // {id, parent id}
847 q.emplace(distinct_id, -1);
848 vis_distinct[distinct_id] = true;
849 while (!q.empty()) {
850 auto [cur_distinct, parent] = q.front();
851 q.pop();
852
853 std::set<int> neigh_distinct;
854 for (int idx : distinct_constraints_[cur_distinct])
855 for (int nxt_distinct :
856 distinct_containing_comp_idx[comp_id[idx]]) {
857 if (nxt_distinct == cur_distinct or
858 nxt_distinct == parent)
859 continue;
860
861 // Cycle found.
862 if (vis_distinct[nxt_distinct])
863 throw error_internal("failed to generate sequence: "
864 "complex constraints");
865
866 neigh_distinct.insert(nxt_distinct);
867 }
868
869 for (int nxt_distinct : neigh_distinct) {
870 vis_distinct[nxt_distinct] = true;
871 q.emplace(nxt_distinct, cur_distinct);
872
873 // Generates this distinct constraint.
874 std::set<T> nxt_defined_values;
875 for (int idx2 : distinct_constraints_[nxt_distinct])
876 if (defined_idx[idx2]) {
877 // There can not be any more defined. This case is
878 // when there are values not coverered by a single
879 // distinct constraint in the tree.
880 if (initially_defined_comp_idx[comp_id[idx2]])
881 throw error_internal(
882 "failed to generate sequence: "
883 "complex constraints");
884
885 nxt_defined_values.insert(vec[idx2]);
886 }
887 int new_value_count =
888 distinct_constraints_[nxt_distinct].size() -
889 static_cast<int>(nxt_defined_values.size());
890 std::vector<T> generated_values = generate_distinct_values(
891 new_value_count, nxt_defined_values);
892 auto val_it = generated_values.begin();
893 for (int idx2 : distinct_constraints_[nxt_distinct])
894 if (!defined_idx[idx2]) {
895 define_comp(comp_id[idx2], *val_it);
896 ++val_it;
897 }
898 }
899 }
900 };
901
902 // Loops through distinct constraints, sorts distinct constraints
903 // by number of defined components (non-increasing). This guarantees
904 // that if there is a valid root (that covers all 'defined'), we find
905 // it.
906 {
907 std::vector<std::pair<int, int>> defined_cnt_and_distinct_idx;
908 int dist_id = 0;
909 for (const std::set<int> &distinct : distinct_constraints_) {
910 int defined_cnt = 0;
911 for (int idx : distinct)
912 if (defined_idx[idx]) {
913 ++defined_cnt;
914 initially_defined_comp_idx[comp_id[idx]] = true;
915 }
916 defined_cnt_and_distinct_idx.emplace_back(defined_cnt, dist_id);
917 ++dist_id;
918 }
919
920 std::sort(defined_cnt_and_distinct_idx.rbegin(),
921 defined_cnt_and_distinct_idx.rend());
922 for (auto [defined_cnt, distinct_idx] :
923 defined_cnt_and_distinct_idx)
924 if (!vis_distinct[distinct_idx])
925 define_tree(distinct_idx);
926 }
927
928 // Loops through distinct constraints do define the rest.
929 for (std::size_t dist_id = 0; dist_id < distinct_constraints_.size();
930 ++dist_id)
931 if (!vis_distinct[dist_id])
932 define_tree(dist_id);
933
934 // Define final values. These values all should be random in [l, r], and
935 // the distinct constraints have already been processed. However, there
936 // can be still equality constraints, so we set entire components.
937 for (int idx = 0; idx < size_; ++idx)
938 if (!defined_idx[idx])
939 define_comp(comp_id[idx], next<T>(value_l_, value_r_));
940
941 if (!values_.empty()) {
942 // Needs to fetch the values from the value set.
943 std::vector<T> value_vec(values_.begin(), values_.end());
944 for (T &val : vec)
945 val = value_vec[val];
946 }
947
948 return instance(vec);
949 }
950};

995 * Permutation generation.
996 *
997 * Permutation are defined always as numbers in [0, n), that is, 0-based.

1000struct permutation : gen_base<permutation> {
1001 int size_; // Size of permutation.
1002 std::vector<std::pair<int, int>> sets; // {idx, value}.
1003
1004 // Creates generator for permutation of size 'size'.
1005 permutation(int size) : size_(size) {
1006 tgen_ensure(size_ > 0, "size must be positive");
1007 }
1008
1009 // Restricts sequences for permutation[idx] = value.
1010 permutation &set(int idx, int value) {
1011 tgen_ensure(0 <= idx and idx < size_, "index must be valid");
1012 sets.emplace_back(idx, value);
1013 return *this;
1014 }
1015
1016 // Permutation instance.
1017 // Operations on an instance are not random.
1018 struct instance {
1019 std::vector<int> vec_; // Permutation.
1020 bool add_1_; // If should add 1, for printing.
1021
1022 instance(const std::vector<int> &vec) : vec_(vec), add_1_(false) {
1023 tgen_ensure(!vec_.empty(), "permutation cannot be empty");
1024 std::vector<bool> vis(vec_.size(), false);
1025 for (std::size_t i = 0; i < vec_.size(); i++) {
1026 tgen_ensure(0 <= vec_[i] and
1027 vec_[i] < static_cast<int>(vec_.size()),
1028 "permutation values must be from `0` to `size-1`");
1029 tgen_ensure(!vis[vec_[i]],
1030 "cannot have repeated values in permutation");
1031 vis[vec_[i]] = true;
1032 }
1033 }
1034 instance(const std::initializer_list<int> &il)
1035 : instance(std::vector<int>(il.begin(), il.end())) {}
1036
1037 // Fetches size.
1038 std::size_t size() const { return vec_.size(); }
1039
1040 // Fetches position idx.
1041 int &operator[](int idx) { return vec_[idx]; }
1042 const int &operator[](int idx) const { return vec_[idx]; }
1043
1044 // Returns parity of the permutation (+1 if even, -1 if odd).
1045 int parity() const {
1046 std::vector<bool> vis(size(), false);
1047 int cycles = 0;
1048
1049 for (std::size_t i = 0; i < size(); ++i)
1050 if (!vis[i]) {
1051 cycles++;
1052 for (int j = i; !vis[j]; j = vec_[j])
1053 vis[j] = true;
1054 }
1055 // Even iff (n - cycles) is even.
1056 return ((size() - cycles) % 2 == 0) ? +1 : -1;
1057 }
1058
1059 // Sorts values in increasign order.
1060 instance &sort() {
1061 std::sort(vec_.begin(), vec_.end());
1062 return *this;
1063 }
1064
1065 // Reverses permutation.
1066 instance &reverse() {
1067 std::reverse(vec_.begin(), vec_.end());
1068 return *this;
1069 }
1070
1071 // Inverse of the permutation.
1072 instance &inverse() {
1073 std::vector<int> inv(size());
1074 for (std::size_t i = 0; i < size(); ++i)
1075 inv[vec_[i]] = i;
1076 swap(vec_, inv);
1077 return *this;
1078 }
1079
1080 // Sets that should print values 1-based.
1081 instance &add_1() {
1082 add_1_ = true;
1083 return *this;
1084 }
1085
1086 // Prints to std::ostream, separated by spaces.
1087 friend std::ostream &operator<<(std::ostream &out,
1088 const instance &inst) {
1089 for (std::size_t i = 0; i < inst.size(); ++i) {
1090 if (i > 0)
1091 out << ' ';
1092 out << inst[i] + inst.add_1_;
1093 }
1094 return out;
1095 }
1096
1097 // Gets a std::vector representing the instance.
1098 std::vector<int> to_std() const { return vec_; }
1099 };
1100
1101 // Generates permutation instance.
1102 instance gen() {
1103 sequence<int> seq(size_, 0, size_ - 1);
1104 seq.distinct();
1105 for (auto [idx, val] : sets)
1106 seq.set(idx, val);
1107 return instance(seq.gen().to_std());
1108 }
1109
1110 // Generates permutation instance, given cycle sizes.
1111 instance gen(std::vector<int> cycle_sizes) {
1112 tgen_ensure(
1113 size_ == std::accumulate(cycle_sizes.begin(), cycle_sizes.end(), 0),
1114 "cycle sizes must add up to size of permutation");
1115
1116 // Creates cycles.
1117 std::vector<int> order(size_);
1118 std::iota(order.begin(), order.end(), 0);
1119 shuffle(order.begin(), order.end());
1120 int idx = 0;
1121 std::vector<std::vector<int>> cycles;
1122 for (int cycle_size : cycle_sizes) {
1123 cycles.emplace_back();
1124 for (int i = 0; i < cycle_size; ++i)
1125 cycles.back().push_back(order[idx++]);
1126 }
1127
1128 // Retrieves permutation from cycles.
1129 std::vector<int> perm(size_, -1);
1130 for (const std::vector<int> &cycle : cycles) {
1131 int cur_size = cycle.size();
1132 for (int i = 0; i < cur_size; ++i)
1133 perm[cycle[i]] = cycle[(i + 1) % cur_size];
1134 }
1135
1136 return instance(perm);
1137 }

1176// O(log^2 n).
1177inline bool is_prime(uint64_t n) {
1178 if (n < 2)
1179 return false;
1180 if (n == 2 or n == 3)
1181 return true;
1182 if (n % 2 == 0)
1183 return false;
1184
1185 uint64_t r = ctzll(n - 1), d = n >> r;
1186 // These bases are guaranteed to work for n <= 2^64.
1187 for (int a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
1188 uint64_t x = expo_mod(a, d, n);
1189 if (x == 1 or x == n - 1 or a % n == 0)
1190 continue;
1191
1192 for (uint64_t j = 0; j < r - 1; ++j) {
1193 x = mul_mod(x, x, n);
1194 if (x == n - 1)
1195 break;
1196 }
1197 if (x != n - 1)

1199 }

1203inline uint64_t pollard_rho_internal(uint64_t n) {
1204 if (n == 1 or is_prime(n))
1205 return n;
1206 auto f = [n](uint64_t x) { return mul_mod(x, x, n) + 1; };
1207
1208 uint64_t x = 0, y = 0, t = 30, prd = 2, x0 = 1, q;
1209 while (t % 40 != 0 or std::gcd(prd, n) == 1) {
1210 if (x == y)
1211 x = ++x0, y = f(x);
1212 q = mul_mod(prd, x > y ? x - y : y - x, n);
1213 if (q != 0)
1214 prd = q;
1215 x = f(x), y = f(f(y)), t++;
1216 }
1217 return std::gcd(prd, n);
1218}

1285// If adjust_distribution = true, then generates uniformly over value ranges.
1286// Otherwise, uniform over primes, so smaller primes are generated more often.
1287inline uint64_t gen_prime(uint64_t l, uint64_t r,
1288 bool adjust_distribution = true) {
1289 if (r < l or r < 2)
1290 throw there_is_no_in_range_error_internal("prime", l, r);
1291 l = std::max<uint64_t>(l, 2);
1292 // There might be no primes in the range.
1293 if (r - l <= 1520) {
1294 std::vector<uint64_t> vals(r - l + 1);
1295 iota(vals.begin(), vals.end(), l);
1296 shuffle(vals.begin(), vals.end());
1297 for (uint64_t i : vals)
1298 if (is_prime(i))
1299 return i;
1300 throw there_is_no_in_range_error_internal("prime", l, r);
1301 }
1302
1303 // Sample with accepting probability proportional to prime density.
1304 while (true) {
1305 uint64_t n = next(l, r);
1306 if (!is_prime(n))
1307 continue;
1308
1309 if (!adjust_distribution or next<double>(0, 1) < log(n) / log(r))
1310 return n;
1311 }

1519inline uint64_t gen_congruent(uint64_t l, uint64_t r,
1520 std::vector<uint64_t> rems,
1521 std::vector<uint64_t> mods) {
1522 if (l > r)
1523 throw there_is_no_in_range_error_internal("congruent number", l, r);
1524 tgen_ensure(rems.size() == mods.size(),
1525 "number of remainders and mods must be the same");
1526
1527 crt_internal crt;
1528 for (int i = 0; i < static_cast<int>(rems.size()); ++i) {
1529 tgen_ensure(rems[i] < mods[i],
1530 "remainder must be smaller than the mod");
1531 crt = crt * crt_internal(rems[i], mods[i]);
1532
1533 if (crt.m > r) {
1534 if (crt.a > r)
1535 throw there_is_no_in_range_error_internal("congruent number", l,
1536 r);
1537
1538 bool valid_candidate = true;
1539 for (int j = 0; j < static_cast<int>(rems.size()); ++j)
1540 if (crt.a % mods[j] != rems[j])
1541 valid_candidate = false;
1542 if (valid_candidate)
1543 return crt.a;
1544 throw there_is_no_in_range_error_internal("congruent number", l, r);
1545 }
1546 }
1547
1548 if (crt.a > r)
1549 throw there_is_no_in_range_error_internal("congruent number", l, r);
1550
1551 uint64_t k_min = crt.a >= l ? 0 : ((l - crt.a) + crt.m - 1) / crt.m;
1552 uint64_t k_max = (r - crt.a) / crt.m;
1553
1554 if (k_min > k_max)
1555 throw there_is_no_in_range_error_internal("congruent number", l, r);
1556
1557 return crt.a + next(k_min, k_max) * crt.m;
1558}

1617// Parition is ordered (composition), that is, (1, 1, 2) != (1, 2, 1).
1618// O(n).
1619// 0 < n.
1620// 0 < part_l.
1621inline std::vector<int> gen_partition(int n, int part_l = 1, int part_r = -1) {
1622 if (part_r == -1)
1623 part_r = n;
1624 tgen_ensure(n > 0 and 0 < part_l and part_l,
1625 "invalid parameters to gen_partition");
1626 tgen_ensure(part_l <= n, "no such partition");
1627
1628 // dp[i] = log(numbers of ways to add to i).
1629 std::vector<long double> dp(n + 1, LOG_ZERO_INTERNAL);
1630 dp[0] = LOG_ONE_INTERNAL;
1631 long double window = LOG_ZERO_INTERNAL;
1632 for (int i = 1; i <= n; ++i) {
1633 if (i >= part_l)
1634 window = add_log_space_internal(window, dp[i - part_l]);
1635 if (i >= part_r + 1)
1636 window = sub_log_space_internal(window, dp[i - part_r - 1]);
1637 dp[i] = window;
1638 }
1639 tgen_ensure(dp[n] != LOG_ZERO_INTERNAL, "partition not found");
1640
1641 // Crazy math tricks ahead.
1642 auto dp_pref = dp;
1643 for (int i = 1; i <= n; i++)
1644 dp_pref[i] = add_log_space_internal(dp_pref[i - 1], dp[i]);
1645
1646 std::vector<int> part;
1647 int sum = n;
1648 while (sum > 0) {
1649 // Will generate a number such that what remains is in [l, r].
1650 int l = std::max(0, sum - part_r), r = sum - part_l;
1651 int nxt_sum = std::min(sum, r);
1652 long double random = next<long double>(0, 1);
1653
1654 // We generate a value X (log space), and then choose nxt_sum such that
1655 // dp_pref[nxt_sum-1] < X <= dp_pref[nxt_sum].
1656
1657 // Math hack:
1658 // Let A = pref[l-1], B = pref[r], U = rand().
1659 // X = log[exp(A) + U * (exp(B) - exp(A))]
1660 // = log{exp(B) * [exp(A) / exp(B) + U * (1 - exp(A) / exp(B))]}
1661 // = B + log[exp(A - B) + U - U * exp(A - B))]
1662 // = B + log[U + (1 - U) * exp(A - B)].
1663 long double val_l = l ? dp_pref[l - 1] : LOG_ZERO_INTERNAL,
1664 val_r = dp_pref[r];
1665 while (nxt_sum > l and
1666 dp_pref[nxt_sum - 1] >=
1667 val_r + log_space_internal(random + (1 - random) *
1668 exp(val_l - val_r)))
1669 --nxt_sum;
1670
1671 part.push_back(sum - nxt_sum);
1672 sum = nxt_sum;
1673 }
1674
1675 return part;
1676}

1678// Parition is ordered (composition), that is, (1, 1, 2) != (1, 2, 1).
1679// O(n) time/memory if part_r = -1, O(n * k) time/memory otherwise.
1680// 0 < k <= n.
1681// 0 <= part_l.
1682inline std::vector<int> gen_partition_fixed_size(int n, int k, int part_l = 0,
1683 int part_r = -1) {
1684 if (part_r == -1)
1685 part_r = n;
1686 tgen_ensure(0 < k and k <= n and 0 <= part_l,
1687 "invalid parameters to gen_partition_fixed_size");
1688 tgen_ensure(static_cast<long long>(k) * part_l <= n and
1689 n <= static_cast<long long>(k) * part_r,
1690 "no such partition");
1691
1692 // What we need to distribute to the parts.
1693 int s = n - k * part_l;
1694
1695 std::vector<int> part(k);
1696 if (part_r == n) {
1697 // Stars and bars - O(n).
1698 std::vector<int> cuts = {-1};
1699
1700 int total = s + k - 1, bars = k - 1;
1701 for (int i = 0; i < total and bars > 0; ++i)
1702 if (next<long double>(0, 1) <
1703 static_cast<long double>(bars) / (total - i)) {
1704 cuts.push_back(i);
1705 --bars;
1706 }
1707 cuts.push_back(total);
1708
1709 // Recovers parts.
1710 for (int i = 0; i < k; ++i)
1711 part[i] = cuts[i + 1] - cuts[i] - 1;
1712 } else {
1713 // DP with log trick - O(nk).
1714 int u = part_r - part_l;
1715
1716 // dp[i][j] = log(#ways to fill i parts with sum j)
1717 std::vector<std::vector<long double>> dp(
1718 k + 1, std::vector<long double>(s + 1, LOG_ZERO_INTERNAL));
1719 dp[0][0] = LOG_ONE_INTERNAL;
1720
1721 for (int i = 1; i <= k; ++i) {
1722 std::vector<long double> pref = dp[i - 1];
1723 for (int j = 1; j <= s; ++j)
1724 pref[j] = add_log_space_internal(pref[j - 1], dp[i - 1][j]);
1725
1726 for (int j = 0; j <= s; ++j) {
1727 dp[i][j] = pref[j];
1728 if (j >= u + 1)
1729 dp[i][j] =
1730 sub_log_space_internal(dp[i][j], pref[j - u - 1]);
1731 }
1732 }
1733
1734 // Recovers parts backwards.
1735 int left_to_distribute = s;
1736 for (int i = k; i >= 1; --i) {
1737 long double log_total = LOG_ZERO_INTERNAL;
1738 for (int j = 0; j <= u and j <= left_to_distribute; ++j)
1739 log_total = add_log_space_internal(
1740 log_total, dp[i - 1][left_to_distribute - j]);
1741 tgen_ensure(log_total != LOG_ZERO_INTERNAL, "partition not found");
1742
1743 // Now we choose a number with probability proportional to
1744 // dp[i-1][.].
1745
1746 // log(rand() * total) = log(rand()) + log(total).
1747 long double random =
1748 log_space_internal(next<long double>(0, 1)) + log_total;
1749
1750 long double cur_prob = LOG_ZERO_INTERNAL;
1751 int chosen = 0;
1752 for (int j = 0; j <= u and j <= left_to_distribute; ++j) {
1753 cur_prob = add_log_space_internal(
1754 cur_prob, dp[i - 1][left_to_distribute - j]);
1755 if (random < cur_prob) {
1756 chosen = j;
1757 break;
1758 }
1759 }
1760
1761 part[k - i] = chosen;
1762 left_to_distribute -= chosen;
1763 }
1764 }
1765
1766 for (int &i : part)
1767 i += part_l;
1768 return part;
1769}
