240// Shuffles [first, last) inplace uniformly, for RandomAccessIterator.
241// O(|container|).
242template <typename It> void shuffle(It first, It last) {
243 if (first == last)
244 return;
245
246 for (It i = first + 1; i != last; ++i)
247 std::iter_swap(i, first + next(0, static_cast<int>(i - first)));
248}

296// Chooses k values from the container, as in a subsequence of size k. Returns a
297// copy.
298// O(|container|).
299template <typename C> C choose(int k, const C &container) {
300 tgen_ensure(0 < k and k <= static_cast<int>(container.size()),
301 "number of elements to choose must be valid");
302 std::vector<typename C::value_type> new_vec;
303 C new_container;
304 int need = k, left = container.size();
305 for (auto cur_it = container.begin(); cur_it != container.end(); ++cur_it) {
306 if (next(1, left--) <= need) {
307 new_container.insert(new_container.end(), *cur_it);
308 need--;
309 }
310 }
311 return new_container;
312}

319template <typename Gen> struct gen_base {
320 // Calls the generator until predicate is true.
321 template <typename Pred, typename... Args>
322 auto gen_until(Pred predicate, int max_tries, Args &&...args) {
323 for (int i = 0; i < max_tries; ++i) {
324 auto inst =
325 static_cast<Gen *>(this)->gen(std::forward<Args>(args)...);
326
327 if (predicate(inst))
328 return inst;
329 }
330
331 throw error_internal("could not generate instance matching predicate");
332 }
333 template <typename Pred, typename T, typename... Args>
334 auto gen_until(Pred predicate, int max_tries, std::initializer_list<T> il,
335 Args &&...args) {
336 return gen_until(predicate, max_tries, std::vector<T>(il),
337 std::forward<Args>(args)...);
338 }

355/*
356 * Opts - options given to the generator.
357 *
358 * Incompatible with testlib.
359 *
360 * Opts are a list of either positional or named options.
361 *
362 * Named options is given in one of the following formats:
363 * 1) -keyname=value or --keyname=value (ex. -n=10 , --test-count=20)
364 * 2) -keyname value or --keyname value (ex. -n 10 , --test-count 20)
365 *
366 * Positional options are number from 0 sequentially.
367 * For example, for "10 -n=20 str" positional option 1 is the string "str".
368 */
369
370inline std::vector<std::string>
371 pos_opts_internal; // Dictionary containing the positional parsed opts.
372inline std::map<std::string, std::string>
373 named_opts_internal; // Global dictionary the named parsed opts.
374
375// Returns true if there is an opt at a given index.
376inline bool has_opt(std::size_t index) {
377 ensure_registered_internal();
378 return 0 <= index and index < pos_opts_internal.size();
379}
380
381// Returns true if there is an opt with a given key.
382inline bool has_opt(const std::string &key) {
383 ensure_registered_internal();
384 return named_opts_internal.count(key) != 0;
385}
386
387template <typename T> T get_opt_internal(const std::string &value) {
388 try {
389 if constexpr (std::is_same_v<T, bool>) {
390 if (value == "true" or value == "1")
391 return true;
392 if (value == "false" or value == "0")
393 return false;
394 } else if constexpr (std::is_integral_v<T>) {
395 if constexpr (std::is_unsigned_v<T>)
396 return static_cast<T>(std::stoull(value));
397 else
398 return static_cast<T>(std::stoll(value));
399 } else if constexpr (std::is_floating_point_v<T>)
400 return static_cast<T>(std::stold(value));
401 else
402 return value; // default: std::string
403 } catch (...) {
404 }
405
406 throw error_internal("invalid value `" + value + "` for type " +
407 typeid(T).name());
408}
409
410// Returns the parsed opt by a given key. If no opts with the given key are
411// found, returns the given default_value.
412template <typename T, typename Key>
413T opt(const Key &key, std::optional<T> default_value = std::nullopt) {
414 ensure_registered_internal();
415 if constexpr (std::is_same_v<Key, int>) {
416 if (!has_opt(key)) {
417 if (default_value)
418 return *default_value;
419 throw error_internal("cannot find key with index " +
420 std::to_string(key));
421 }
422 return get_opt_internal<T>(pos_opts_internal[key]);
423 } else { // std::string
424 if (!has_opt(key)) {
425 if (default_value)
426 return *default_value;
427 throw error_internal("cannot find key with key " +
428 std::string(key));
429 }
430 return get_opt_internal<T>(named_opts_internal[key]);
431 }
432}

434inline void parse_opts_internal(int argc, char **argv) {
435 // Parses the opts into `pos_opts_internal` vector and `named_opts_internal`
436 // map. Starting from 1 to ignore the name of the executable.
437 for (int i = 1; i < argc; ++i) {
438 std::string key(argv[i]);
439
440 if (key[0] == '-') {
441 tgen_ensure(key.size() > 1,
442 "invalid opt (" + std::string(argv[i]) + ")");
443 if ('0' <= key[1] and key[1] <= '9') {
444 // This case is a positional negative number argument
445 pos_opts_internal.push_back(key);
446 continue;
447 }
448
449 // pops first char '-'
450 key = key.substr(1);
451 } else {
452 // This case is a positional argument that does not start with '-'
453 pos_opts_internal.push_back(key);
454 continue;
455 }
456
457 // Pops a possible second char '-'.
458 if (key[0] == '-') {
459 tgen_ensure(key.size() > 1,
460 "invalid opt (" + std::string(argv[i]) + ")");
461
462 // pops first char '-'
463 key = key.substr(1);
464 }
465
466 // Assumes that, if it starts with '-' and second char is not a digit,
467 // then it is a <key, value> pair.
468 // 1 or 2 chars '-' have already been poped.
469
470 std::size_t eq = key.find('=');
471 if (eq != std::string::npos) {
472 // This is the '--key=value' case.
473 std::string value = key.substr(eq + 1);
474 key = key.substr(0, eq);
475 tgen_ensure(!key.empty() and !value.empty(),
476 "expected non-empty key/value in opt (" +
477 std::string(argv[1]));
478 tgen_ensure(named_opts_internal.count(key) == 0,
479 "cannot have repeated keys");
480 named_opts_internal[key] = value;
481 } else {
482 // This is the '--key value' case.
483 tgen_ensure(named_opts_internal.count(key) == 0,
484 "cannot have repeated keys");
485 tgen_ensure(argv[i + 1], "value cannot be empty");
486 named_opts_internal[key] = std::string(argv[i + 1]);
487 ++i;
488 }
489 }
490}
491inline void set_seed_internal(int argc, char **argv) {
492 std::vector<uint32_t> seed;
493
494 // Starting from 1 to ignore the name of the executable.
495 for (int i = 1; i < argc; ++i) {
496 // We append the number of chars, and then the list of chars.
497 int size_pos = seed.size();
498 seed.push_back(0);
499 for (char *s = argv[i]; *s != '\0'; ++s) {
500 ++seed[size_pos];
501 seed.push_back(*s);
502 }
503 }
504 std::seed_seq seq(seed.begin(), seed.end());
505 rng_internal.seed(seq);
506}

512 pos_opts_internal.clear();
513 named_opts_internal.clear();
514 parse_opts_internal(argc, argv);
515
516 registered_internal = true;
517}

529template <typename T> struct sequence : gen_base<sequence<T>> {
530 int size_; // Size of sequence.
531 T value_l_, value_r_; // Range of defined values.
532 std::set<T> values_; // Set of values. If empty, use range. if not,
533 // represents the possible values, and the range
534 // represents the index in this set)
535 std::map<T, int>
536 value_idx_in_set_; // Index of every value in the set above.
537 std::vector<std::pair<T, T>> val_range_; // Range of values of each index.
538 std::vector<std::vector<int>> neigh_; // Adjacency list of equality.
539 std::vector<std::set<int>>
540 distinct_constraints_; // All distinct constraints.
541
542 // Creates generator for sequences of size 'size', with random T in [l, r].
543 sequence(int size, T value_l, T value_r)
544 : size_(size), value_l_(value_l), value_r_(value_r), neigh_(size) {
545 tgen_ensure(size_ > 0, "size must be positive");
546 tgen_ensure(value_l_ <= value_r_, "value range must be valid");
547 for (int i = 0; i < size_; ++i)
548 val_range_.emplace_back(value_l_, value_r_);
549 }
550
551 // Creates sequence with value set.
552 sequence(int size, std::set<T> values)
553 : size_(size), values_(values), neigh_(size) {
554 tgen_ensure(size_ > 0, "size must be positive");
555 tgen_ensure(!values.empty(), "value set must be non-empty");
556 value_l_ = 0, value_r_ = values.size() - 1;
557 for (int i = 0; i < size_; ++i)
558 val_range_.emplace_back(value_l_, value_r_);
559 int idx = 0;
560 for (T value : values_)
561 value_idx_in_set_[value] = idx++;
562 }
563
564 // Restricts sequences for sequence[idx] = value.
565 sequence &set(int idx, T value) {
566 tgen_ensure(0 <= idx and idx < size_, "index must be valid");
567 if (values_.size() == 0) {
568 auto &[left, right] = val_range_[idx];
569 if (left == right and value_l_ != value_r_) {
570 tgen_ensure(left == value,
571 "must not set to two different values");
572 } else {
573 tgen_ensure(left <= value and value <= right,
574 "value must be in the defined range");
575 }
576 left = right = value;
577 } else {
578 tgen_ensure(values_.count(value),
579 "value must be in the set of values");
580 auto &[left, right] = val_range_[idx];
581 int new_val = value_idx_in_set_[value];
582 tgen_ensure(left <= new_val and new_val <= right,
583 "must not set to two different values");
584 left = right = new_val;
585 }
586 return *this;
587 }
588
589 // Restricts sequences for sequence[idx_1] = sequence[idx_2].
590 sequence &equal(int idx_1, int idx_2) {
591 tgen_ensure(0 <= std::min(idx_1, idx_2) and
592 std::max(idx_1, idx_2) < size_,
593 "indices must be valid");
594 if (idx_1 == idx_2)
595 return *this;
596
597 neigh_[idx_1].push_back(idx_2);
598 neigh_[idx_2].push_back(idx_1);
599 return *this;
600 }
601
602 // Restricts sequences for sequence[left..right] to have all equal values.
603 sequence &equal_range(int left, int right) {
604 tgen_ensure(0 <= left and left <= right and right < size_,
605 "range indices bust be valid");
606 for (int i = left; i < right; ++i)
607 equal(i, i + 1);
608 return *this;
609 }
610
611 // Restricts sequences for sequence[S] to be distinct, for given subset S of
612 // indices.
613 // You can not add two of these restrictions with intersection.
614 sequence &distinct(std::set<int> indices) {
615 if (!indices.empty())
616 distinct_constraints_.push_back(indices);
617 return *this;
618 }
619
620 // Restricts sequences for sequence[idx_1] != sequence[idx_2].
621 sequence &different(int idx_1, int idx_2) {
622 std::set<int> indices = {idx_1, idx_2};
623 return distinct(indices);
624 }
625
626 // Restricts sequences with distinct elements.
627 sequence &distinct() {
628 std::vector<int> indices(size_);
629 std::iota(indices.begin(), indices.end(), 0);
630 return distinct(std::set<int>(indices.begin(), indices.end()));
631 }

688 // Generates a uniformly random list of k distinct values in `[value_l,
689 // value_r]`, such that no value is in `forbidden_values`.
690 std::vector<T>
691 generate_distinct_values(int k, const std::set<T> &forbidden_values) {
692 for (auto forbidden : forbidden_values)
693 tgen_ensure(value_l_ <= forbidden and forbidden <= value_r_);
694 // We generate our numbers in the range [0, num_available) with
695 // num_available = (r-l+1)-(forbidden_values.size()), and then map them
696 // to the correct range. We will run k steps of Fisher–Yates, using a
697 // map to store a virtual sequence that starts with a[i] = i.
698 T num_available = (value_r_ - value_l_ + 1) - forbidden_values.size();
699 if (num_available < k)
700 throw error_internal(
701 "failed to generate sequence: complex constraints");
702 std::map<T, T> virtual_list;
703 std::vector<T> gen_list;
704 for (int i = 0; i < k; ++i) {
705 T j = next<T>(i, num_available - 1);
706 T vj = virtual_list.count(j) ? virtual_list[j] : j;
707 T vi = virtual_list.count(i) ? virtual_list[i] : i;
708
709 virtual_list[j] = vi, virtual_list[i] = vj;
710
711 gen_list.push_back(virtual_list[i]);
712 }
713
714 // Shifts back to correct range, but there might still be values
715 // that we can not use.
716 for (T &value : gen_list)
717 value += value_l_;
718
719 // Now for every generated value, we shift it by how many forbidden
720 // values are <= to it.
721 std::vector<std::pair<T, int>> values_sorted;
722 for (std::size_t i = 0; i < gen_list.size(); ++i)
723 values_sorted.emplace_back(gen_list[i], i);
724 // We iterate through them in increasing order.
725 std::sort(values_sorted.begin(), values_sorted.end());
726 auto cur_it = forbidden_values.begin();
727 int smaller_forbidden_count = 0;
728 for (auto [val, idx] : values_sorted) {
729 while (cur_it != forbidden_values.end() and
730 *cur_it <= val + smaller_forbidden_count)
731 ++cur_it, ++smaller_forbidden_count;
732 gen_list[idx] += smaller_forbidden_count;
733 }
734
735 return gen_list;
736 }
737
738 // Generates sequence instance.
739 // O(n log n).
740 instance gen() {
741 std::vector<T> vec(size_);
742 std::vector<bool> defined_idx(
743 size_, false); // For every index, if it has been set in `vec`.
744
745 std::vector<int> comp_id(size_, -1); // Component id of each index.
746 std::vector<std::vector<int>> comp(size_); // Component of each comp-id.
747 int comp_count = 0; // Number of different components.
748
749 // Defines value of entire component.
750 auto define_comp = [&](int cur_comp, T val) {
751 for (int idx : comp[cur_comp]) {
752 tgen_ensure(!defined_idx[idx]);
753 vec[idx] = val;
754 defined_idx[idx] = true;
755 }
756 };
757
758 // Groups = components.
759 {
760 std::vector<bool> vis(size_, false); // Visited for each index.
761 for (int idx = 0; idx < size_; ++idx)
762 if (!vis[idx]) {
763 T new_value;
764 bool value_defined = false;
765
766 // BFS to visit the connected component, grouping equal
767 // values.
768 std::queue<int> q({idx});
769 vis[idx] = true;
770 std::vector<int> component;
771 while (!q.empty()) {
772 int cur_idx = q.front();
773 q.pop();
774
775 component.push_back(cur_idx);
776
777 // Checks value.
778 auto [l, r] = val_range_[cur_idx];
779 if (l == r) {
780 if (!value_defined) {
781 // We found the value.
782 value_defined = true;
783 new_value = l;
784 } else if (new_value != l) {
785 // We found a contradiction
786 throw contradiction_error_internal(
787 "sequence",
788 "tried to set value to `" +
789 std::to_string(new_value) +
790 "`, but it was already set as `" +
791 std::to_string(l) + "`");
792 }
793 }
794
795 for (int nxt_idx : neigh_[cur_idx]) {
796 if (!vis[nxt_idx]) {
797 vis[nxt_idx] = true;
798 q.push(nxt_idx);
799 }
800 }
801 }
802
803 // Group entire component, checking if value is defined.
804 for (int cur_idx : component) {
805 comp_id[cur_idx] = comp_count;
806 comp[comp_id[cur_idx]].push_back(cur_idx);
807 }
808
809 // Sets value if needed.
810 if (value_defined)
811 define_comp(comp_count, new_value);
812
813 ++comp_count;
814 }
815 }
816
817 // Initial parsing of distinct constraints.
818 std::vector<std::set<int>> distinct_containing_comp_idx(comp_count);
819 {
820 int dist_id = 0;
821 for (const std::set<int> &distinct : distinct_constraints_) {
822 // Checks if there are too many distinct values.
823 if (static_cast<uint64_t>(distinct.size() - 1) +
824 static_cast<uint64_t>(value_l_) >
825 static_cast<uint64_t>(value_r_))
826 throw contradiction_error_internal(
827 "sequence",
828 "tried to generate " + std::to_string(distinct.size()) +
829 " distinct values, but the maximum is " +
830 std::to_string(value_r_ - value_l_ + 1));
831
832 // Checks if two values in same component are marked as
833 // different.
834 std::set<int> comp_ids;
835 for (int idx : distinct) {
836 if (comp_ids.count(comp_id[idx]))
837 throw contradiction_error_internal(
838 "sequence", "tried to set two indices as equal and "
839 "different");
840 comp_ids.insert(comp_id[idx]);
841
842 distinct_containing_comp_idx[comp_id[idx]].insert(dist_id);
843 }
844 ++dist_id;
845 }
846 }
847
848 // If some value is in >= 3 sets, then there is a cycle.
849 for (auto &distinct_containing : distinct_containing_comp_idx)
850 if (distinct_containing.size() >= 3)
851 throw error_internal(
852 "failed to generate sequence: complex constraints");
853
854 std::vector<bool> vis_distinct(distinct_constraints_.size(), false);
855 std::vector<bool> initially_defined_comp_idx(comp_count, false);
856
857 // Fills the value in a tree defined by distinct constraints.
858 auto define_tree = [&](int distinct_id) {
859 // The set `distinct_constraints_[distinct_id]` can have some values
860 // that are defined.
861
862 // Generates set of already defined values.
863 std::set<T> defined_values;
864 for (int idx : distinct_constraints_[distinct_id])
865 if (defined_idx[idx]) {
866 // Checks if two values in `distinct_constraints_[dist_id]`
867 // have been set to the same value
868 if (defined_values.count(vec[idx]))
869 throw contradiction_error_internal(
870 "sequence",
871 "tried to set two indices as equal and different");
872
873 defined_values.insert(vec[idx]);
874 }
875
876 // Generates values in this root distinct constraint.
877 {
878 int new_value_count =
879 distinct_constraints_[distinct_id].size() -
880 static_cast<int>(defined_values.size());
881 std::vector<T> generated_values =
882 generate_distinct_values(new_value_count, defined_values);
883 auto val_it = generated_values.begin();
884 for (int idx : distinct_constraints_[distinct_id])
885 if (defined_idx[idx]) {
886 // The root can cover these components, but there should
887 // not be any other defined in this tree.
888 initially_defined_comp_idx[comp_id[idx]] = false;
889 } else {
890 define_comp(comp_id[idx], *val_it);
891 ++val_it;
892 }
893 }
894
895 // BFS on the tree of distinct constraints.
896 std::queue<std::pair<int, int>> q; // {id, parent id}
897 q.emplace(distinct_id, -1);
898 vis_distinct[distinct_id] = true;
899 while (!q.empty()) {
900 auto [cur_distinct, parent] = q.front();
901 q.pop();
902
903 std::set<int> neigh_distinct;
904 for (int idx : distinct_constraints_[cur_distinct])
905 for (int nxt_distinct :
906 distinct_containing_comp_idx[comp_id[idx]]) {
907 if (nxt_distinct == cur_distinct or
908 nxt_distinct == parent)
909 continue;
910
911 // Cycle found.
912 if (vis_distinct[nxt_distinct])
913 throw error_internal("failed to generate sequence: "
914 "complex constraints");
915
916 neigh_distinct.insert(nxt_distinct);
917 }
918
919 for (int nxt_distinct : neigh_distinct) {
920 vis_distinct[nxt_distinct] = true;
921 q.emplace(nxt_distinct, cur_distinct);
922
923 // Generates this distinct constraint.
924 std::set<T> nxt_defined_values;
925 for (int idx2 : distinct_constraints_[nxt_distinct])
926 if (defined_idx[idx2]) {
927 // There can not be any more defined. This case is
928 // when there are values not coverered by a single
929 // distinct constraint in the tree.
930 if (initially_defined_comp_idx[comp_id[idx2]])
931 throw error_internal(
932 "failed to generate sequence: "
933 "complex constraints");
934
935 nxt_defined_values.insert(vec[idx2]);
936 }
937 int new_value_count =
938 distinct_constraints_[nxt_distinct].size() -
939 static_cast<int>(nxt_defined_values.size());
940 std::vector<T> generated_values = generate_distinct_values(
941 new_value_count, nxt_defined_values);
942 auto val_it = generated_values.begin();
943 for (int idx2 : distinct_constraints_[nxt_distinct])
944 if (!defined_idx[idx2]) {
945 define_comp(comp_id[idx2], *val_it);
946 ++val_it;
947 }
948 }
949 }
950 };
951
952 // Loops through distinct constraints, sorts distinct constraints
953 // by number of defined components (non-increasing). This guarantees
954 // that if there is a valid root (that covers all 'defined'), we find
955 // it.
956 {
957 std::vector<std::pair<int, int>> defined_cnt_and_distinct_idx;
958 int dist_id = 0;
959 for (const std::set<int> &distinct : distinct_constraints_) {
960 int defined_cnt = 0;
961 for (int idx : distinct)
962 if (defined_idx[idx]) {
963 ++defined_cnt;
964 initially_defined_comp_idx[comp_id[idx]] = true;
965 }
966 defined_cnt_and_distinct_idx.emplace_back(defined_cnt, dist_id);
967 ++dist_id;
968 }
969
970 std::sort(defined_cnt_and_distinct_idx.rbegin(),
971 defined_cnt_and_distinct_idx.rend());
972 for (auto [defined_cnt, distinct_idx] :
973 defined_cnt_and_distinct_idx)
974 if (!vis_distinct[distinct_idx])
975 define_tree(distinct_idx);
976 }
977
978 // Loops through distinct constraints do define the rest.
979 for (std::size_t dist_id = 0; dist_id < distinct_constraints_.size();
980 ++dist_id)
981 if (!vis_distinct[dist_id])
982 define_tree(dist_id);
983
984 // Define final values. These values all should be random in [l, r], and
985 // the distinct constraints have already been processed. However, there
986 // can be still equality constraints, so we set entire components.
987 for (int idx = 0; idx < size_; ++idx)
988 if (!defined_idx[idx])
989 define_comp(comp_id[idx], next<T>(value_l_, value_r_));
990
991 if (!values_.empty()) {
992 // Needs to fetch the values from the value set.
993 std::vector<T> value_vec(values_.begin(), values_.end());
994 for (T &val : vec)
995 val = value_vec[val];
996 }
997
998 return instance(vec);
999 }
1000};

1022// Chooses k values from the sequence, as in a subsequence of size k.

1034 }

1047/*
1048 * Permutation generation.
1049 *
1050 * Permutation are defined always as numbers in [0, n), that is, 0-based.
1051 */
1052
1053struct permutation : gen_base<permutation> {
1054 int size_; // Size of permutation.
1055 std::vector<std::pair<int, int>> sets; // {idx, value}.
1056
1057 // Creates generator for permutation of size 'size'.
1058 permutation(int size) : size_(size) {
1059 tgen_ensure(size_ > 0, "size must be positive");
1060 }
1061
1062 // Restricts sequences for permutation[idx] = value.
1063 permutation &set(int idx, int value) {
1064 tgen_ensure(0 <= idx and idx < size_, "index must be valid");
1065 sets.emplace_back(idx, value);
1066 return *this;
1067 }
1068
1069 // Permutation instance.
1070 // Operations on an instance are not random.
1071 struct instance {
1072 std::vector<int> vec_; // Permutation.
1073 bool add_1_; // If should add 1, for printing.
1074
1075 instance(const std::vector<int> &vec) : vec_(vec), add_1_(false) {
1076 tgen_ensure(!vec_.empty(), "permutation cannot be empty");
1077 std::vector<bool> vis(vec_.size(), false);
1078 for (int i = 0; i < size(); ++i) {
1079 tgen_ensure(0 <= vec_[i] and
1080 vec_[i] < static_cast<int>(vec_.size()),
1081 "permutation values must be from `0` to `size-1`");
1082 tgen_ensure(!vis[vec_[i]],
1083 "cannot have repeated values in permutation");
1084 vis[vec_[i]] = true;
1085 }
1086 }
1087 instance(const std::initializer_list<int> &il)
1088 : instance(std::vector<int>(il.begin(), il.end())) {}
1089
1090 // Fetches size.
1091 int size() const { return vec_.size(); }
1092
1093 // Fetches position idx.
1094 int &operator[](int idx) { return vec_[idx]; }
1095 const int &operator[](int idx) const { return vec_[idx]; }
1096
1097 // Returns parity of the permutation (+1 if even, -1 if odd).
1098 // O(n).
1099 int parity() const {
1100 std::vector<bool> vis(size(), false);
1101 int cycles = 0;
1102
1103 for (int i = 0; i < size(); ++i)
1104 if (!vis[i]) {
1105 cycles++;
1106 for (int j = i; !vis[j]; j = vec_[j])
1107 vis[j] = true;
1108 }
1109 // Even iff (n - cycles) is even.
1110 return ((size() - cycles) % 2 == 0) ? +1 : -1;
1111 }
1112
1113 // Sorts values in increasign order.
1114 // O(n).
1115 instance &sort() {
1116 for (int i = 0; i < size(); ++i)
1117 vec_[i] = i;
1118 return *this;
1119 }
1120
1121 // Reverses permutation.
1122 // O(n).
1123 instance &reverse() {
1124 std::reverse(vec_.begin(), vec_.end());
1125 return *this;
1126 }
1127
1128 // Inverse of the permutation.
1129 // O(n).
1130 instance &inverse() {
1131 std::vector<int> inv(size());
1132 for (int i = 0; i < size(); ++i)
1133 inv[vec_[i]] = i;
1134 swap(vec_, inv);
1135 return *this;
1136 }
1137
1138 // Sets that should print values 1-based.
1139 // O(1).
1140 instance &add_1() {
1141 add_1_ = true;
1142 return *this;
1143 }
1144
1145 // Prints to std::ostream, separated by spaces.
1146 friend std::ostream &operator<<(std::ostream &out,
1147 const instance &inst) {
1148 for (int i = 0; i < inst.size(); ++i) {
1149 if (i > 0)
1150 out << ' ';
1151 out << inst[i] + inst.add_1_;
1152 }
1153 return out;
1154 }
1155
1156 // Gets a std::vector representing the instance.
1157 std::vector<int> to_std() const { return vec_; }
1158 };
1159
1160 // Generates permutation instance.
1161 // O(n log n).
1162 instance gen() {
1163 // TODO: generate directly in O(n), instead of calling sequence<int>.
1164 sequence<int> seq(size_, 0, size_ - 1);
1165 seq.distinct();
1166 for (auto [idx, val] : sets)
1167 seq.set(idx, val);
1168 return instance(seq.gen().to_std());
1169 }
1170
1171 // Generates permutation instance, given cycle sizes.
1172 // O(n).
1173 instance gen(std::vector<int> cycle_sizes) {
1174 tgen_ensure(
1175 size_ == std::accumulate(cycle_sizes.begin(), cycle_sizes.end(), 0),
1176 "cycle sizes must add up to size of permutation");
1177
1178 // Creates cycles.
1179 std::vector<int> order(size_);
1180 std::iota(order.begin(), order.end(), 0);
1181 shuffle(order.begin(), order.end());
1182 int idx = 0;
1183 std::vector<std::vector<int>> cycles;
1184 for (int cycle_size : cycle_sizes) {
1185 cycles.emplace_back();
1186 for (int i = 0; i < cycle_size; ++i)
1187 cycles.back().push_back(order[idx++]);
1188 }
1189
1190 // Retrieves permutation from cycles.
1191 std::vector<int> perm(size_, -1);
1192 for (const std::vector<int> &cycle : cycles) {
1193 int cur_size = cycle.size();
1194 for (int i = 0; i < cur_size; ++i)
1195 perm[cycle[i]] = cycle[(i + 1) % cur_size];
1196 }
1197
1198 return instance(perm);
1199 }

1221inline uint64_t mul_mod_internal(uint64_t a, uint64_t b, uint64_t m) {
1222 return static_cast<unsigned __int128>(a) * b % m;
1223}

1225// O(log n).
1226// 0 <= x < m.
1227inline uint64_t expo_mod_internal(uint64_t x, uint64_t y, uint64_t m) {
1228 if (!y)
1229 return 1;
1230 uint64_t ans = expo_mod_internal(mul_mod_internal(x, x, m), y / 2, m);
1231 return y % 2 ? mul_mod_internal(x, ans, m) : ans;
1232}

1234// O(log^2 n).
1235inline bool is_prime(uint64_t n) {
1236 if (n < 2)
1237 return false;
1238 if (n == 2 or n == 3)
1239 return true;
1240 if (n % 2 == 0)
1241 return false;
1242
1243 uint64_t r = ctzll_internal(n - 1), d = n >> r;
1244 // These bases are guaranteed to work for n <= 2^64.
1245 for (int a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
1246 uint64_t x = expo_mod_internal(a, d, n);
1247 if (x == 1 or x == n - 1 or a % n == 0)
1248 continue;
1249
1250 for (uint64_t j = 0; j < r - 1; ++j) {
1251 x = mul_mod_internal(x, x, n);
1252 if (x == n - 1)
1253 break;
1254 }
1255 if (x != n - 1)
1256 return false;
1257 }
1258 return true;
1259}

1261inline uint64_t pollard_rho_internal(uint64_t n) {
1262 if (n == 1 or is_prime(n))
1263 return n;
1264 auto f = [n](uint64_t x) { return mul_mod_internal(x, x, n) + 1; };
1265
1266 uint64_t x = 0, y = 0, t = 30, prd = 2, x0 = 1, q;
1267 while (t % 40 != 0 or std::gcd(prd, n) == 1) {
1268 if (x == y)
1269 x = ++x0, y = f(x);
1270 q = mul_mod_internal(prd, x > y ? x - y : y - x, n);
1271 if (q != 0)
1272 prd = q;
1273 x = f(x), y = f(f(y)), t++;
1274 }
1275 return std::gcd(prd, n);
1276}

1278inline std::vector<uint64_t> factor_internal(uint64_t n) {
1279 if (n == 1)
1280 return {};
1281 if (is_prime(n))
1282 return {n};
1283 uint64_t d = pollard_rho_internal(n);
1284 std::vector<uint64_t> l = factor_internal(d), r = factor_internal(n / d);
1285 l.insert(l.end(), r.begin(), r.end());
1286 return l;
1287}

1314// O(log mod).
1315// 0 < a < mod.
1316// gcd(a, mod) = 1.
1317inline __int128 modular_inverse_128_internal(__int128 a, __int128 mod) {
1318 tgen_ensure(0 < a and a < mod,
1319 "remainder must be positive and smaller than the mod");
1320
1321 __int128 t = 0, new_t = 1;
1322 __int128 r = mod, new_r = a;
1323
1324 while (new_r != 0) {
1325 __int128 q = r / new_r;
1326
1327 auto tmp_t = t - q * new_t;
1328 t = new_t;
1329 new_t = tmp_t;
1330
1331 auto tmp_r = r - q * new_r;
1332 r = new_r;
1333 new_r = tmp_r;
1334 }
1335
1336 tgen_ensure(r == 1, "remainder and mod must be coprime");
1337
1338 if (t < 0)
1339 t += mod;
1340 return t;
1341}

1362// O(log r) expected.
1363// If adjust_distribution = true, then generates uniformly over value ranges.
1364// Otherwise, uniform over primes, so smaller primes are generated more often.
1365inline uint64_t gen_prime(uint64_t l, uint64_t r,
1366 bool adjust_distribution = true) {
1367 if (r < l or r < 2)
1368 throw there_is_no_in_range_error_internal("prime", l, r);
1369 l = std::max<uint64_t>(l, 2);
1370 // There might be no primes in the range.
1371 if (r - l <= 1520) {

1373 iota(vals.begin(), vals.end(), l);
1374 shuffle(vals.begin(), vals.end());

1376 if (is_prime(i))
1377 return i;
1378 throw there_is_no_in_range_error_internal("prime", l, r);

1382 while (true) {

1385 continue;

1387 if (!adjust_distribution or next<double>(0, 1) < log(n) / log(r))
1388 return n;
1389 }

1654// O(|mods| + log r).
1655// |rems| = |mods|.
1656// rems_i < mods_i.
1657inline uint64_t gen_congruent(uint64_t l, uint64_t r,
1658 std::vector<uint64_t> rems,
1659 std::vector<uint64_t> mods) {
1660 if (l > r)
1661 throw there_is_no_in_range_error_internal("congruent number", l, r);
1662 tgen_ensure(rems.size() == mods.size(),
1663 "number of remainders and mods must be the same");
1664
1665 crt_internal crt;
1666 for (int i = 0; i < static_cast<int>(rems.size()); ++i) {
1667 tgen_ensure(rems[i] < mods[i],
1668 "remainder must be smaller than the mod");
1669 crt = crt * crt_internal(rems[i], mods[i]);
1670
1671 if (crt.a == -1 or crt.m > r) {
1672 if (!(l <= crt.a and crt.a <= r))
1673 throw there_is_no_in_range_error_internal("congruent number", l,
1674 r);
1675
1676 bool valid_candidate = true;
1677 for (int j = 0; j < static_cast<int>(rems.size()); ++j)
1678 if (crt.a % mods[j] != rems[j])
1679 valid_candidate = false;
1680 if (valid_candidate)
1681 return crt.a;
1682 throw there_is_no_in_range_error_internal("congruent number", l, r);
1683 }
1684 }
1685
1686 uint64_t k_min = crt.a >= l ? 0 : ((l - crt.a) + crt.m - 1) / crt.m;
1687 uint64_t k_max = (r - crt.a) / crt.m;
1688
1689 if (k_min > k_max)
1690 throw there_is_no_in_range_error_internal("congruent number", l, r);
1691
1692 return crt.a + next(k_min, k_max) * crt.m;
1693}

1758// Parition is ordered (composition), that is, (1, 1, 2) != (1, 2, 1).
1759// O(n).
1760// 0 < n.
1761// 0 < part_l.
1762inline std::vector<int> gen_partition(int n, int part_l = 1, int part_r = -1) {
1763 if (part_r == -1)
1764 part_r = n;
1765 part_r = std::min(part_r, n);
1766 tgen_ensure(n > 0 and part_l > 0, "invalid parameters to gen_partition");
1767 tgen_ensure(part_l <= n and part_r > 0, "no such partition");
1768
1769 // dp[i] = log(numbers of ways to add to i).
1770 std::vector<long double> dp(n + 1, LOG_ZERO_INTERNAL);
1771 dp[0] = LOG_ONE_INTERNAL;
1772 long double window = LOG_ZERO_INTERNAL;
1773 for (int i = 1; i <= n; ++i) {
1774 if (i >= part_l)
1775 window = add_log_space_internal(window, dp[i - part_l]);
1776 if (i >= part_r + 1)
1777 window = sub_log_space_internal(window, dp[i - part_r - 1]);
1778 dp[i] = window;
1779 }
1780 tgen_ensure(dp[n] >= 0, "no such partition");

1782 // Crazy math tricks ahead.
1783 auto dp_pref = dp;
1784 for (int i = 1; i <= n; ++i)
1785 dp_pref[i] = add_log_space_internal(dp_pref[i - 1], dp[i]);
1786
1787 std::vector<int> part;
1788 int sum = n;
1789 while (sum > 0) {
1790 // Will generate a number such that what remains is in [l, r].
1791 int l = std::max(0, sum - part_r), r = sum - part_l;
1792 tgen_ensure_against_bug(r >= 0);
1793
1794 int nxt_sum = std::min(sum, r);
1795 long double random = next<long double>(0, 1);
1796

1798 // dp_pref[nxt_sum-1] < X <= dp_pref[nxt_sum].
1799
1800 // Math hack:
1801 // Let A = pref[l-1], B = pref[r], U = rand().
1802 // X = log[exp(A) + U * (exp(B) - exp(A))]
1803 // = log{exp(B) * [exp(A) / exp(B) + U * (1 - exp(A) / exp(B))]}
1804 // = B + log[exp(A - B) + U - U * exp(A - B))]
1805 // = B + log[U + (1 - U) * exp(A - B)].
1806 long double val_l = l ? dp_pref[l - 1] : LOG_ZERO_INTERNAL,
1807 val_r = dp_pref[r];
1808 while (nxt_sum > l and
1809 dp_pref[nxt_sum - 1] >=
1810 val_r + log_space_internal(random + (1 - random) *
1811 exp(val_l - val_r)))
